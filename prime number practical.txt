 class Program1
{
  
  public void PrimeNo()
{
   int num=  5;

    int count=0;


    if(num<2)
{
        Console.WriteLine("number is not prime");

}

else
{
       for(int i=1;i<=num;i++)
          {
             if(num%i==0)
          {
             count++;
         }

        }

      if(count==2)
{
    Console.WriteLine("number is prime");
}
else
{

Console.WriteLine("number is not prime");
}


}



}


}
